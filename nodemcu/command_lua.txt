local f = file.open("init.lua", "r")
if f then
    repeat
        local chunk = file.read(128, f)
        if chunk then
            uart.write(0, chunk)
            tmr.delay(50000)
        end
    until not chunk
    f:close()
    print("\n--- TRANSFERENCIA FINALIZADA ---")
else
    print("Error: No se pudo abrir el archivo")
end

# Ejemplo de blink
pin = 4
estado = gpio.LOW
gpio.mode(pin, gpio.OUTPUT)
miTemporizador = tmr.create()
miTemporizador:register(1000, tmr.ALARM_AUTO, function()
	if estado == gpio.LOW then
		estado = gpio.HIGH)
	else
		estado = gpio.LOW
	end
	gpio.write(pin, estado)
	print("LED en estado: "..estado)
end)
miTemporizador:start()
